19:   const files = Object.assign(tracked(root, diagnostics), additionalFiles);
20:   // The recipe is input even before initial adoption has staged it.
21:   files['edition.json'] ??= {data: fs.readFileSync(path.join(root, 'edition.json')).toString('base64'), executable: false};
22:   const components = [];
23:   for (const component of recipe.components) {
24:     const selected = component.source;

26:     const reference = selected.kind === 'git' ? selected : selected.origin;
27:     const baseline = reference ? snapshot(root, reference, retain) : null;
28:     if (selected.kind === 'git') merge(files, selected.destination, baseline.files);
29:     else for (const name of selected.paths) {

31:     }
